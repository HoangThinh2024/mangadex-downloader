SEARCH VỚI POWERERSHELL:
────────────────────────
$search = Invoke-RestMethod "http://localhost:8000/api/search/manga" `
  -Body @{ title = "One Piece"; status = "ongoing"; limit = 10 }
$search.data | Format-Table title, status, year

DOWNLOAD VỚI PYTHON:
──────────────────
import requests

res = requests.post("http://localhost:8000/api/url/download", json={
    "url": "https://mangadex.org/title/...",
    "save_as": "pdf",
    "chapters": "1-10"
})

job_id = res.json()["job_id"]
print(f"Download started: {job_id}")
